const rowContainer = document.getElementById('metal-rows');
const volumeInput = document.getElementById('target-volume');

function buildRows() {
    for (let i = 1; i <= 5; i++) {
        const row = document.createElement('div');
        row.className = 'metal-row';

        const select = document.createElement('select');
        select.id = `metal-${i}`;
        select.setAttribute('aria-label', `Metal ${i}`);

        metalNames.forEach(name => {
            const option = document.createElement('option');
            option.value = name;
            option.textContent = name;
            select.appendChild(option);
        });

        const percentageField = document.createElement('div');
        percentageField.className = 'percentage-field';

        const percentage = document.createElement('input');
        percentage.id = `percentage-${i}`;
        percentage.type = 'number';
        percentage.min = '0';
        percentage.max = '100';
        percentage.step = 'any';
        percentage.value = '0';
        percentage.setAttribute('aria-label', `Metal ${i} percentage`);

        const symbol = document.createElement('span');
        symbol.className = 'percent-symbol';
        symbol.textContent = '%';

        percentageField.append(percentage, symbol);
        row.append(select, percentageField);
        rowContainer.appendChild(row);

        select.addEventListener('change', calculate);
        percentage.addEventListener('input', calculate);
    }
}

function calculate() {
    const totalVolume = Math.max(0, Number(volumeInput.value) || 0);
    let totalPercentage = 0;
    const entries = [];
    
let totalWeight = 0;    

    for (let i = 1; i <= 5; i++) {
        const metal = document.getElementById(`metal-${i}`).value;
        const input = document.getElementById(`percentage-${i}`);
        const percentage = Math.max(0, Number(input.value) || 0);

        if (percentage > 100) {
            // Keep the entered value visible, but flag it as invalid below.
        }

        if (metal !== 'None' && percentage > 0) {
            totalPercentage += percentage;
const metalVolume = totalVolume * percentage / 100;
const metalWeight = metalVolume * metalDensities[metal];

totalWeight += metalWeight;

entries.push({
    metal,
    percentage,
    volume: metalVolume,
    weight: metalWeight
});
        }
    }

    document.getElementById('display-volume').textContent =
        totalVolume.toFixed(2);
    document.getElementById('display-percentage').textContent =
        totalPercentage.toFixed(2) + '%';
    document.getElementById('remaining-volume').textContent =
        (totalVolume * (1 - totalPercentage / 100)).toFixed(2);
        
// NEW CODE: Display total weight and overall density
const overallDensity = totalVolume > 0
    ? totalWeight / totalVolume
    : 0;

document.getElementById('display-weight').textContent =
    totalWeight.toFixed(2);

document.getElementById('display-density').textContent =
    overallDensity.toFixed(2);

// Existing code continues below        
    const status = document.getElementById('percentage-status');
    const anyInvalid = Array.from(
        document.querySelectorAll('[id^="percentage-"]')
    ).some(input => {
        const p = Number(input.value);
        return p < 0 || p > 100;
    });

    if (anyInvalid || totalPercentage > 100) {
        status.textContent = 'Percentages exceed the allowed total. Please check your entries.';
        status.className = 'error';
    } else if (totalPercentage === 100) {
        status.textContent = 'Percentages total 100% — ready to craft.';
        status.className = 'good';
    } else if (totalPercentage > 0) {
        status.textContent = 'Percentages must total 100%. Some volume is unallocated.';
        status.className = 'warning';
    } else {
        status.textContent = 'Enter metal percentages.';
        status.className = 'warning';
    }

    const breakdown = document.getElementById('breakdown');
    breakdown.replaceChildren();

    entries.forEach(entry => {
        const li = document.createElement('li');
        const name = document.createElement('span');
        const amount = document.createElement('strong');

        name.textContent = `${entry.metal} (${entry.percentage.toFixed(2)}%)`;
        amount.textContent = `${entry.volume.toFixed(2)} volume`;
        li.append(name, amount);
        breakdown.appendChild(li);
    });

    if (entries.length === 0) {
        const li = document.createElement('li');
        li.textContent = 'Select a metal and enter its percentage.';
        breakdown.appendChild(li);
    }
}

volumeInput.addEventListener('input', calculate);

document.getElementById('clear-button').addEventListener('click', () => {
    volumeInput.value = '100';

    for (let i = 1; i <= 5; i++) {
        document.getElementById(`metal-${i}`).value = 'None';
        document.getElementById(`percentage-${i}`).value = '0';
    }

    calculate();
});

buildRows();
calculate();
